cmake_minimum_required(VERSION 3.16)
project(YtDlpGui VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets Network Concurrent)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets Network Concurrent)
add_executable(ytdlp-gui WIN32 main.cpp downloadlibrary.h downloadqueue.h mediapreview.h mediaformats.h playlist.h usbstorage.h fileoperations.h filemanager.h settingssidebar.h uilanguage.h)
target_link_libraries(ytdlp-gui PRIVATE Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Network Qt${QT_VERSION_MAJOR}::Concurrent)
if(UNIX AND NOT APPLE)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS DBus)
    target_link_libraries(ytdlp-gui PRIVATE Qt${QT_VERSION_MAJOR}::DBus)
endif()
install(TARGETS ytdlp-gui RUNTIME DESTINATION bin)

option(YTDLP_GUI_BUILD_TESTS "Build deterministic offline integration checks" OFF)
if(YTDLP_GUI_BUILD_TESTS)
    enable_testing()
    add_executable(fake-engine tests/fake_engine.cpp)
    target_link_libraries(fake-engine PRIVATE Qt${QT_VERSION_MAJOR}::Core)
    add_executable(gui-checks tests/checks.cpp)
    target_link_libraries(gui-checks PRIVATE Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Network Qt${QT_VERSION_MAJOR}::Concurrent)
    if(UNIX AND NOT APPLE)
        target_link_libraries(gui-checks PRIVATE Qt${QT_VERSION_MAJOR}::DBus)
    endif()
    add_test(NAME gui-checks COMMAND gui-checks $<TARGET_FILE:fake-engine>)
    set_tests_properties(gui-checks PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;LC_ALL=ru_RU.UTF-8" TIMEOUT 30)
endif()
